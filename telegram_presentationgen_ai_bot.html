Серверный код Telegram Бота (Python)

Этот код позволяет развернуть **реального Telegram бота** на сервере с помощью бибилиотек python-telegram-bot и новейшей google-genai.

1. Установка зависимостей:

pip install python-telegram-bot google-genai pillow
import os
import json
import asyncio
from telegram import Update, InlineKeyboardButton, InlineKeyboardMarkup, WebAppInfo
from telegram.ext import Application, CommandHandler, MessageHandler, filters, ContextTypes
from google import genai
from google.genai import types

# Инициализация API ключей
GEMINI_API_KEY = os.getenv("GEMINI_API_KEY", "YOUR_GEMINI_API_KEY")
BOT_TOKEN = os.getenv("BOT_TOKEN", "YOUR_TELEGRAM_BOT_TOKEN")

# Клиент Gemini API
client = genai.Client(api_key=GEMINI_API_KEY)

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    welcome_text = (
        "👋 Привет! Я AI-генератор презентаций.\n\n"
        "Отправь мне тему для презентации, например:\n"
        "• «Будущее искусственного интеллекта в 2026 году»\n"
        "• «План маркетинговой стратегии»\n\n"
        "Или открой наше встроенное WebApp приложение:"
    )
    keyboard = [
        [InlineKeyboardButton("🚀 Открыть WebApp Генератор", web_app=WebAppInfo(url="https://your-domain.com"))]
    ]
    reply_markup = InlineKeyboardMarkup(keyboard)
    await update.message.reply_text(welcome_text, reply_markup=reply_markup)

async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE):
    topic = update.message.text
    msg = await update.message.reply_text("⏳ Генерирую структуру презентации с помощью Gemini AI...")

    prompt = f"""
    Создай структуру презентации на тему: "{topic}".
    Количество слайдов: 5.
    Верни результат strictly в формате JSON со следующей структурой:
    {{
      "title": "Заголовок презентации",
      "slides": [
        {{
          "slide_number": 1,
          "title": "Заголовок слайда",
          "subtitle": "Подзаголовок",
          "bullets": ["Тезис 1", "Тезис 2", "Тезис 3"],
          "image_prompt": "Промпт для генерации картинки на английском"
        }}
      ]
    }}
    """

    try:
        response = client.models.generate_content(
            model='gemini-2.5-flash',
            contents=prompt,
            config=types.GenerateContentConfig(response_mime_type="application/json")
        )
        slides_data = json.loads(response.text)
        
        summary = f"✨ **Презентация создана:** {slides_data.get('title')}\n\n"
        for slide in slides_data.get('slides', []):
            summary += f"📌 **Слайд {slide['slide_number']}: {slide['title']}**\n"
            for bullet in slide.get('bullets', []):
                summary += f" • {bullet}\n"
            summary += "\n"

        await msg.edit_text(summary, parse_mode="Markdown")

    except Exception as e:
        await msg.edit_text(f"❌ Ошибка генерации: {str(e)}")

def main():
    app = Application.builder().token(BOT_TOKEN).build()
    app.add_handler(CommandHandler("start", start))
    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_message))
    print("Бот успешно запущен!")
    app.run_polling()

if __name__ == "__main__":
    main()
